APIs that work everywhere.

Telepact bridges programs across any transport boundary. JSON in, JSON out — with optional binary efficiency, dynamic response shaping, and type‑safe code generation.

Core Innovation

Three pillars, zero compromises

Telepact combines the best ideas from REST, gRPC, and GraphQL into one cohesive ecosystem — without their trade-offs.

{ }

JSON as a Query Language

API calls and SELECT-style field selection achieved entirely through JSON abstractions. No special parsers, no DSLs. Clients using only native JSON libraries are first-class citizens.

⚡

Binary Without Code Gen

Binary protocols established through runtime handshakes — not build-time code generation. Opt into MessagePack efficiency without any mandatory toolchain setup.

🔗

Hypermedia Without HTTP

API responses include "links" — functions with pre-filled arguments. HATEOAS-style navigation using pure JSON, over any transport. Discoverability built in.

- fn.divide:                     # Define a function
    x: "integer"                 # Required argument
    y: "integer"
  ->:                            # Return type is a union
    - Ok_:
        result: "number"
    - ErrorCannotDivideByZero: {}

- fn.getUsers:
    page!: "integer"             # Optional argument (note the !)
    limit!: "integer"
  ->:
    - Ok_:

        total: "integer"

- struct.User:                   # Define a reusable struct
    id: "integer"
    name: "string"
    email: "string"

from telepact import Message, Server, TelepactSchema, TelepactSchemaFiles

files = TelepactSchemaFiles('./api')
schema = TelepactSchema.from_file_json_map(files.filenames_to_json)

async def handler(request_message: Message) -> Message:
    fn = request_message.get_body_target()
    args = request_message.body[fn]

    if fn == 'fn.divide':
        x, y = args['x'], args['y']
        if y == 0:
            return Message({}, {'ErrorCannotDivideByZero': {}})
        return Message({}, {'Ok_': {'result': x / y}})

    raise Exception(f'Unknown function: {fn}')

options = Server.Options()
options.auth_required = False
server = Server(schema, handler, options)

# Plug into any framework — Starlette, Flask, FastAPI...
async def http_handler(request):
    response = await server.process(await request.body())
    return Response(content=response.bytes)
import { Client, ClientOptions, Message, Serializer } from 'telepact';

// Pluggable transport — use fetch, WebSocket, NATS, anything
const adapter = async (m: Message, s: Serializer): Promise<Message> => {
    const res = await fetch('/api/telepact', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: s.serialize(m),
    });
    return s.deserialize(new Uint8Array(await res.arrayBuffer()));
};

const client = new Client(adapter, new Serializer(), new ClientOptions());

// Type-safe call with dynamic field selection
const response = await client.call('fn.divide', { x: 10, y: 3 });
console.log(response.body);
// { "Ok_": { "result": 3.333... } }
package main

import (
    "github.com/telepact/telepact/lib/go/pkg/telepact"
)

func main() {
    files := telepact.NewTelepactSchemaFiles("./api")
    schema := telepact.NewTelepactSchemaFromFileJsonMap(files.FilenamesToJson)

    handler := func(msg *telepact.Message) *telepact.Message {
        fn := msg.GetBodyTarget()
        args := msg.Body[fn].(map[string]any)

        switch fn {
        case "fn.divide":
            x := args["x"].(float64)
            y := args["y"].(float64)
            if y == 0 {
                return telepact.NewMessage(
                    map[string]any{},
                    map[string]any{"ErrorCannotDivideByZero": map[string]any{}},
                )
            }
            return telepact.NewMessage(
                map[string]any{},
                map[string]any{"Ok_": map[string]any{"result": x / y}},
            )
        }
        return nil
    }

    options := telepact.DefaultServerOptions()
    options.AuthRequired = false
    server := telepact.NewServer(schema, handler, options)
    // Mount server.Process to any HTTP handler
}
import io.github.telepact.Message;
import io.github.telepact.Server;
import io.github.telepact.TelepactSchema;
import io.github.telepact.TelepactSchemaFiles;

var files = new TelepactSchemaFiles("./api");
var schema = TelepactSchema.fromFileJsonMap(files.getFilenamesToJson());

Server.Handler handler = (requestMessage) -> {
    var fn = requestMessage.getBodyTarget();
    var args = (Map<String, Object>) requestMessage.getBody().get(fn);

    if ("fn.divide".equals(fn)) {
        int x = (int) args.get("x");
        int y = (int) args.get("y");
        if (y == 0) {
            return new Message(Map.of(), Map.of("ErrorCannotDivideByZero", Map.of()));
        }
        return new Message(Map.of(), Map.of("Ok_", Map.of("result", (double) x / y)));
    }
    throw new RuntimeException("Unknown function: " + fn);
};

var options = new Server.Options();
options.setAuthRequired(false);
var server = new Server(schema, handler, options);
